```ts twoslash
// @noErrors
import { cascadeOf, defaultBreakpoints, mediaQuery } from 'puck-kit';

cascadeOf(defaultBreakpoints, 'mobile').map((b) => b.key); // ['base', 'tablet', 'mobile']
mediaQuery(defaultBreakpoints[3]); // '(max-width:767px)'
```
